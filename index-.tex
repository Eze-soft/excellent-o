    <!-- BOOKMARKING LOGIC ENGINE -->
    <script>
        let savedJobs = JSON.parse(localStorage.getItem("eo_saved_jobs") || "[]");

        function updateSavedCount() {
            const countEl = document.getElementById("saved-count");
            if (countEl) countEl.innerText = savedJobs.length;
        }

        function toggleBookmark(jobTitle, jobLocation) {
            const index = savedJobs.findIndex(j => j.title === jobTitle);
            if (index > -1) {
                savedJobs.splice(index, 1);
            } else {
                savedJobs.push({ title: jobTitle, location: jobLocation });
            }
            localStorage.setItem("eo_saved_jobs", JSON.stringify(savedJobs));
            updateSavedCount();
            renderSavedJobs();
        }

        function renderSavedJobs() {
            const container = document.getElementById("saved-jobs-container");
            if (!container) return;

            if (savedJobs.length === 0) {
                container.innerHTML = "<p style='color: var(--dark-bg); font-size: 0.75rem; margin: 2px 0;'>No saved roles yet. Click the ⭐ icon on any vacancy card above to bookmark it for quick access.</p>";
                return;
            }

            container.innerHTML = "";
            savedJobs.forEach(job => {
                const card = document.createElement("div");
                card.className = "job-card";
                card.style.padding = "0.5rem";
                const cleanTitle = job.title.replace(/'/g, "\\'");
                card.innerHTML = `
                    <div>
                        <span class="visa-badge" style="font-size:0.55rem; padding:1px 4px;">SAVED</span>
                        <h3 style="font-size:0.78rem; margin:2px 0;">${job.title}</h3>
                        <p class="location" style="font-size:0.68rem; margin:0;">📍 ${job.location}</p>
                    </div>
                    <button onclick="toggleBookmark('${cleanTitle}', '${job.location}')" class="apply-btn" style="color: #c0392b; border:none; background:none; text-align:left; padding:0; font-size:0.68rem;">Remove ❌</button>
                `;
                container.appendChild(card);
            });
        }
    </script>

        function fetchLiveJob(requestUrl) {
            const container = document.getElementById("seasonal-jobs-container");
            if (!container) return;

            container.innerHTML = "<p>Loading live jobs...</p>";

            // Use a default search term if the input element isn't found
            const keywordInput = document.getElementById("seasonal-keyword");
            const searchTerm = keywordInput ? keywordInput.value || "job" : "job";

            const requestedUrl = typeof requestUrl === "string" ? requestUrl.trim() : "";
            const serverOrigin = window.location.port === "5500"
                ? "http://localhost:3000"
                : window.location.origin;
            const url = requestedUrl && requestedUrl !== "undefined"
                ? requestedUrl
                : `${serverOrigin}/api/live-jobs?keyword=${encodeURIComponent(searchTerm)}`;

            fetch(url)
                .then(response => {
                    if (!response.ok) throw new Error("Live job feed temporarily unavailable");
                    return response.json();
                })
                .then(data => {
                    container.innerHTML = "";
                    if (data.results && data.results.length > 0) {
                        data.results.forEach(job => {
                            container.innerHTML += `
                                <div class="job-card" style="border: 1px solid #ccc; padding: 15px; margin-bottom: 10px; border-radius: 8px;">
                                    <h3>${job.title}</h3>
                                    <p><strong>Company:</strong> ${job.company.display_name}</p>
                                    <p><strong>Location:</strong> ${job.location.display_name}</p>
                                    <a href="${job.redirect_url}" target="_blank" style="color: blue; text-decoration: underline;">View & Apply</a>
                                </div>
                            `;
                        });
                    } else {
                        container.innerHTML = "<p>No live jobs found.</p>";
                    }
                })
                .catch(error => {
                    console.error("Error fetching live jobs:", error);
                    container.innerHTML = "<p>Failed to load live jobs. Please check your connection.</p>";
                });
        }

        async function searchLocalJobs() {
            const container = document.getElementById("local-jobs-container");
            if (!container) return;
            const keyword = document.getElementById("local-keyword").value.trim();
            const city = document.getElementById("local-city").value.trim();
            const countrySelect = document.getElementById("local-country-select");
            const country = countrySelect ? countrySelect.value || "gb" : "gb";

            container.innerHTML = "<p style='color: var(--dark-bg); font-weight: bold; font-size: 0.75rem;'>Fetching country listings...</p>";

            const serverOrigin = window.location.port === "5500"
                ? "http://localhost:3000"
                : window.location.origin;
            let url = `${serverOrigin}/api/live-jobs?country=${encodeURIComponent(country)}&keyword=${encodeURIComponent(keyword || "job")}`;
            if (city) url += `&city=${encodeURIComponent(city)}`;

            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), 10000);

            try {
                const response = await fetch(url, { signal: controller.signal });
                if (!response.ok) throw new Error("Live job feed temporarily unavailable");
                const data = await response.json();
                container.innerHTML = "";

                if (data.results && data.results.length > 0) {
                    renderLocalCards(data.results, container);
                } else {
                    container.innerHTML = "<p style='color: var(--dark-bg); font-weight: bold; font-size: 0.75rem;'>No active vacancies found for this country selection.</p>";
                }
            } catch (err) {
                const message = err.name === "AbortError"
                    ? "The live jobs feed took too long to respond."
                    : "The live jobs feed is temporarily unavailable.";
                container.innerHTML = `<p style='color: #c0392b; font-weight: bold; font-size: 0.75rem;'>${message} Please try again.</p><button type="button" class="mini-btn" onclick="searchLocalJobs()">Retry</button>`;
            } finally {
                clearTimeout(timeoutId);
            }
        }

        async function handleEmployerPosting(e) {
            e.preventDefault();

            var company = document.getElementById('emp_company_name').value;
            var title = document.getElementById('emp_job_title').value;
            var location = document.getElementById('emp_job_location').value;
            var visaType = document.getElementById('emp_visa_type').value;

            var serverOrigin = window.location.port === "5500"
                ? "http://localhost:3000"
                : window.location.origin;

            try {
                var response = await fetch(`${serverOrigin}/api/post-job`, {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({
                        title: title,
                        company: company,
                        contactEmail: document.getElementById('emp_contact_email').value,
                        location: location,
                        visaType: visaType,
                        description: description,
                        featured: true
                    })
                });

                var result = await response.json();

                alert("Unable to post this job. Please make sure the Node server is running with npm start, then try again.");
            }

        async function loadFeaturedJobs() {
            var serverOrigin = window.location.port === "5500"
                ? "http://localhost:3000"
                : window.location.origin;

                var response = await fetch(`${serverOrigin}/api/featured-jobs`);
                if (!response.ok) throw new Error("Failed to load featured jobs");

                var jobs = await response.json();
                var container = document.getElementById('featured-jobs-section');
                if (!container) return;

                container.innerHTML = '';
                if (jobs.length === 0) {
                    container.innerHTML = '<p>No featured jobs right now.</p>';
                    return;

                jobs.forEach(function (job) {
                    var title = job.title || job.jobTitle || 'Featured opportunity';
                    var company = job.company || job.companyName || 'Featured employer';

        let currentIndex = 0;

        function rotateSponsor() {
            currentIndex = (currentIndex + 1) % sponsors.length;
            const currentSponsor = sponsors[currentIndex];
            const sponsorImg = document.getElementById("sponsorImg");
            const sponsorLink = document.getElementById("sponsorLink");
            if (sponsorImg && sponsorLink) {
                sponsorImg.src = currentSponsor.image;
                sponsorLink.href = currentSponsor.url;
            }
        }

        function startRotation() {
            clearInterval(rotationInterval);
            rotationInterval = setInterval(rotateSponsor, 5000);
        }

        function stopRotation() {
            clearInterval(rotationInterval);
        }

        const adContainer = document.querySelector(".sponsor-ad-container");
        if (adContainer) {
            adContainer.addEventListener("mouseenter", stopRotation);
            adContainer.addEventListener("mouseleave", startRotation);
            adContainer.addEventListener("click", stopRotation);
        }

        startRotation();
